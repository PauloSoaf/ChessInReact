function apply(p: FastPosition, s: number, m: number): boolean { if (s) p.cast(s); if (p.make(m)) return true; if (s) p.uncast(); return false; }
function undo(p: FastPosition, s: number) { p.unmake(); if (s) p.uncast(); }
/** EVERY legal turn of the side to move: no pruning at all. */
function everyTurn(p: FastPosition): [number, number][] {
  const out: [number, number][] = p.legalMoves().map(m => [SPELL_NONE, m] as [number, number]);
  const spells: number[] = [];
  if (p.canCast(SPELL_FREEZE)) for (let c = 0; c < 64; c++) spells.push(encodeSpell(SPELL_FREEZE, c));
  if (p.canCast(SPELL_JUMP)) for (let c = 0; c < 64; c++) if (p.board[c]) spells.push(encodeSpell(SPELL_JUMP, c));
  for (const s of spells) { p.cast(s); const ms = p.legalMoves(); p.uncast(); for (const m of ms) out.push([s, m]); }
  return out;
}

const deadline = Date.now() + 500000;
const proven = new Map<string, boolean>();
/** White to move: some attacker turn wins within n attacker moves, against EVERY defence. */
function wins(p: FastPosition, n: number, path: string[]): boolean {
  if (p.findKingCapture()) return true;
  if (n <= 1) return false;
  if (Date.now() > deadline) throw new Error('tempo esgotado em ' + path.join(' '));
  const key = p.keyLo + ':' + p.keyHi + ':' + n;
  const memo = proven.get(key); if (memo !== undefined) return memo;
  const hint = search.search(p, { timeMs: 1e9, maxDepth: n === 2 ? 3 : 5 });
  const ordered = [...hint.rootTurns].sort((a, b) => Number(b === hint.best) - Number(a === hint.best) || b.score - a.score);
  let ok = false; let tried = 0;
  for (const turn of ordered) {
    if (++tried > (n === 2 ? 400 : 12)) break;
    if (!apply(p, turn.spell, turn.move)) continue;
    ok = allRepliesLose(p, n - 1, [...path, name(turn.spell, turn.move)]);
    undo(p, turn.spell);
    if (ok) break;
  }
  if (!ok && path.length > failure.length) failure = [...path, '(brancas nao tem mate em ' + (n - 1) + ')'];
  proven.set(key, ok); return ok;
}
function allRepliesLose(p: FastPosition, n: number, path: string[]): boolean {
  if (p.findKingCapture()) return false;
  const replies = everyTurn(p);
  if (!replies.length) return false;
  for (const [s, m] of replies) {
    if (!apply(p, s, m)) continue;
    nodes++;
    const ok = wins(p, n, [...path, name(s, m)]);
    undo(p, s);
    if (!ok) return false;
  }
  return true;
}
